index.js · 角色选择 + 自动登录态

onLoad() {
  if (auth.isLoggedIn()) {
    const role = auth.getUser().current_role
    if (ROLE_META[role]) wx.switchTab({ url: ROLE_META[role].page })
  }
},

onPickRole(e) {
  const role = e.currentTarget.dataset.role
  const doEnter = () => auth.switchRole(role).then(() => wx.switchTab({ url: ROLE_META[role].page }))
  if (!auth.isLoggedIn()) {
    auth.login().then(() => auth.bindRole(role)).then(doEnter)
  } else {
    const user = auth.getUser()
    (user.roles || []).includes(role) ? doEnter() : auth.bindRole(role).then(doEnter)
  }
}
